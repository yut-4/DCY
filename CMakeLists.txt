cmake_minimum_required(VERSION 3.20)
project(dcy VERSION 0.1 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

find_package(SQLite3 REQUIRED)
find_package(OpenSSL REQUIRED)
find_path(LIBCLANG_INCLUDE_DIR clang-c/Index.h REQUIRED)
find_library(LIBCLANG_LIBRARY NAMES clang libclang REQUIRED)
option(DCY_ENABLE_MLPACK "Build experimental mlpack lexical-neighbor retrieval" OFF)

add_executable(dcy src/main.cpp)
target_include_directories(dcy PRIVATE ${LIBCLANG_INCLUDE_DIR})
target_link_libraries(dcy PRIVATE SQLite3::SQLite3 OpenSSL::Crypto ${LIBCLANG_LIBRARY})
target_compile_options(dcy PRIVATE -Wall -Wextra -Wpedantic)
if(DCY_ENABLE_MLPACK)
  find_package(Armadillo REQUIRED)
  find_package(cereal REQUIRED)
  find_package(ensmallen REQUIRED)
  find_package(mlpack REQUIRED)
  target_compile_definitions(dcy PRIVATE DCY_ENABLE_MLPACK=1)
  target_include_directories(dcy PRIVATE ${ARMADILLO_INCLUDE_DIRS} ${CEREAL_INCLUDE_DIR} ${ENSMALLEN_INCLUDE_DIR} ${MLPACK_INCLUDE_DIR})
  target_link_libraries(dcy PRIVATE ${ARMADILLO_LIBRARIES})
endif()

enable_testing()
add_test(NAME dcy_smoke COMMAND ${CMAKE_COMMAND} -DDCY=$<TARGET_FILE:dcy> -DROOT=${CMAKE_CURRENT_SOURCE_DIR} -P ${CMAKE_CURRENT_SOURCE_DIR}/tests/smoke.cmake)
add_test(NAME dcy_invalidation COMMAND ${CMAKE_COMMAND} -DDCY=$<TARGET_FILE:dcy> -DROOT=${CMAKE_CURRENT_SOURCE_DIR} -P ${CMAKE_CURRENT_SOURCE_DIR}/tests/invalidation.cmake)
find_package(Python3 COMPONENTS Interpreter QUIET)
if(Python3_Interpreter_FOUND)
  add_test(NAME dcy_mcp_smoke COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/mcp_smoke.py $<TARGET_FILE:dcy>)
endif()
